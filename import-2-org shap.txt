import shap
import pandas as pd
from sklearn.ensemble import RandomForestClassifier

def rm_main(data):
    # 1. Scheid de features van de voorspelde labels
    # We halen 'Survived' (label) en de door RapidMiner gemaakte voorspellingskolommen weg
    columns_to_drop = ['Survived', 'prediction(Survived)']
    # Verwijder ook eventuele confidence-kolommen als die er zijn (bijv. confidence(Survived))
    columns_to_drop = [col for col in columns_to_drop if col in data.columns]
    
    X = data.drop(columns=columns_to_drop)
    
    # Omdat het model in RapidMiner leeft, trainen we in Python heel snel een 
    # schaduwmodel op exact dezelfde data om de SHAP-waarden te berekenen.
    # Dit is de standaardprocedure voor tabellen in de Execute Python-operator.
    y = data['Survived']
    
    # Zorg dat tekstkolommen (zoals Gender/Port) snel numeriek worden voor de sklearn Random Forest
    X_encoded = pd.get_dummies(X, drop_first=True)
    
    shadow_model = RandomForestClassifier(n_estimators=50, random_state=42)
    shadow_model.fit(X_encoded, y)
    
    # 2. Bereken SHAP met de TreeExplainer
    explainer = shap.TreeExplainer(shadow_model)
    shap_values = explainer(X_encoded)
    
    # 3. Converteer naar DataFrame met 'shap_' prefix
    shap_df = pd.DataFrame(shap_values.values, columns=[f"shap_{col}" for col in X_encoded.columns])
    
    # 4. Plak de SHAP-waarden vast aan de originele dataset
    return pd.concat([data.reset_index(drop=True), shap_df], axis=1)
